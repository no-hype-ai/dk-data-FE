# DK Molecule Data Platform — Pipeline Schedule Reference
# =====================================================
#
# NOTE: The production scheduler runs as a long-lived Docker service
# (pipeline_scheduler.py), NOT as a cron daemon. This file documents
# the equivalent cron expressions for reference and for manual/CI runs.
#
# To run manually:
#   python -m dk_data.services.data_platform.sync_runner \
#       --sources clinicaltrials,openfda_labels --tier daily
#
# Environment variables (set via Doppler / .env):
#   DATABASE_URL          PostgreSQL connection string
#   NCBI_API_KEY          PubMed / NCBI E-utilities (10 req/s authenticated)
#   OPENALEX_API_KEY      OpenAlex (required since Feb 2026)
#   EPO_CONSUMER_KEY      EPO Open Patent Services OAuth2
#   EPO_CONSUMER_SECRET
#   ORCID_CLIENT_ID       ORCID public API OAuth2
#   ORCID_CLIENT_SECRET
#   SEC_USER_AGENT        SEC EDGAR fair-access user-agent string
#   OPENFDA_API_KEY       openFDA (optional, raises rate limit)
#
# Log files: /var/log/dk-data/cron_<tier>.log
# Manifests: data/raw/.manifests/<source>.json (ETag + cursor state)
#
# =====================================================

PATH=/usr/local/bin:/usr/bin:/bin
DK_HOME=/app

# DAILY SOURCES — 02:00 UTC
# High-change-frequency sources updated every day
# =====================================================

# =====================================================
# SQLMESH TRANSFORMATIONS — after ingestion tiers
# =====================================================

# Run bronze → silver → gold transforms after daily ingestion

# Full SQLMesh run on Sundays after weekly ingestion

# =====================================================
# LOG ROTATION (Weekly on Sunday at midnight)
# =====================================================

0 0 * * 0 find /var/log/dk-data -name "*.log" -mtime +30 -exec gzip {} \;
0 0 * * 0 find /var/log/dk-data -name "*.gz" -mtime +90 -delete
