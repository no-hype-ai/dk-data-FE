  name: fetch-[SOURCE_NAME_DASHED]

    app.kubernetes.io/name: fetch-[SOURCE_NAME_DASHED]
    app.kubernetes.io/component: ingestion
    app.kubernetes.io/part-of: dk-data
spec:
  schedule: "[CRON_SCHEDULE]"
  concurrencyPolicy: Forbid
  successfulJobsHistoryLimit: 3
  failedJobsHistoryLimit: 3
  jobTemplate:
    spec:
      backoffLimit: 2
      activeDeadlineSeconds: [TIMEOUT_SECONDS]

        metadata:

          restartPolicy: OnFailure
          containers:
            - name: fetch-[SOURCE_NAME_DASHED]
              image: ghcr.io/data-kinetic/dk-data-fe/job-trigger:latest
              command:
                - python
                - -m
                - dk_data.ingestion.fetch_data
                - --source
                - [SOURCE_NAME]
                - --data-dir
                - /data/raw

                      key: POSTGRES_USER

                  valueFrom:
                    secretKeyRef:
                      name: dk-data-secrets
                      key: POSTGRES_PASSWORD

                    secretKeyRef:
